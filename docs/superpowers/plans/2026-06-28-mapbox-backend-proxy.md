# Mapbox Backend Proxy — Geocoding & Static Map Images Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Mapbox-shaped frontend API calls with stable, provider-agnostic backend endpoints for geocoding and static map images, with server-side caching and browser cache-breaking via a `updatedAt` timestamp on Location.

**Architecture:** A new `MapController` exposes `/api/map/geocode/suggest`, `/api/map/geocode/reverse`, `/api/map/image/location/{id}`, and `/api/map/image/event/{id}`. Geocoding delegates to `GeocoderInterface` (implemented by `BazingaGeocoderAdapter` wrapping geocoder-php's BazingaGeocoderBundle with the Mapbox provider). Map images delegate to `MapImageProviderInterface` (implemented by `MapboxMapImageProvider` via Symfony HttpClient). The `Location` entity gains a `HasUpdatedAt` trait; its Unix timestamp is appended to image URLs as `?v=` to bust both browser and server cache on coordinate changes.

**Tech Stack:** PHP 8.4, Symfony 8.1, Doctrine ORM, `geocoder-php/bazinga-geocoder-bundle`, `geocoder-php/mapbox-provider`, Symfony Cache (filesystem pool), Vite + React + TypeScript.

## Global Constraints

- PHP 8.4 strict types on every file (`declare(strict_types=1);`)
- Entity IDs: UUID v7 via `Uuid::v7()->toRfc4122()`, never auto-increment
- All backend tests: PHPUnit WebTestCase, real MySQL test DB on port 3360, `DatabaseResetter::reset()` in `setUp()`
- Frontend: Vite SPA, no Next.js patterns, `import.meta.env.VITE_API_URL` for backend URL
- No comments except where the WHY is non-obvious

```php
<?php
declare(strict_types=1);

namespace App\Service\Map;

use Geocoder\Geocoder;
use Geocoder\Model\AddressInterface;
use Geocoder\Query\GeocodeQuery;
use Geocoder\Query\ReverseQuery;

class BazingaGeocoderAdapter implements GeocoderInterface
{
    public function __construct(private readonly Geocoder $geocoder) {}

    public function suggest(string $query, ?float $proximityLat = null, ?float $proximityLng = null, int $limit = 6): array
    {
        $geoQuery = GeocodeQuery::create($query)->withLimit($limit);
        if ($proximityLat !== null && $proximityLng !== null) {
            $geoQuery = $geoQuery->withData('proximity', sprintf('%f,%f', $proximityLng, $proximityLat));
        }

        $collection = $this->geocoder->geocodeQuery($geoQuery);
        $results    = [];

        foreach ($collection as $address) {
            $coords = $address->getCoordinates();
            if ($coords === null) {
                continue;
            }
            $results[] = new SuggestionResult(
                label:   $this->buildLabel($address),
                lat:     $coords->getLatitude(),
                lng:     $coords->getLongitude(),
                street:  $this->buildStreet($address),
                zip:     $address->getPostalCode(),
                city:    $address->getLocality(),
                country: $address->getCountry()?->getName(),
            );
        }

        return $results;
    }

    public function reverse(float $lng, float $lat): ?GeocodedAddress
    {
        $collection = $this->geocoder->reverseQuery(ReverseQuery::fromCoordinates($lat, $lng));

        if ($collection->isEmpty()) {
            return null;
        }

        $address = $collection->first();

        return new GeocodedAddress(
            label:   $this->buildLabel($address),
            street:  $this->buildStreet($address),
            zip:     $address->getPostalCode(),
            city:    $address->getLocality(),
            country: $address->getCountry()?->getName(),
        );
    }

    private function buildLabel(AddressInterface $address): string
    {
        $parts = array_filter([
            $this->buildStreet($address),
            $address->getLocality(),
            $address->getCountry()?->getName(),
        ]);

        return implode(', ', $parts);
    }

    private function buildStreet(AddressInterface $address): ?string
    {
        $name = $address->getStreetName();
        if ($name === null) {
            return null;
        }
        $number = $address->getStreetNumber();

        return $number !== null ? "$name $number" : $name;
    }
}
```

```php
<?php
declare(strict_types=1);

namespace App\EventListener;

use App\Entity\Location;
use Doctrine\ORM\Event\PostUpdateEventArgs;
use Doctrine\ORM\Event\PreUpdateEventArgs;
use Doctrine\ORM\Events;
use Symfony\Bridge\Doctrine\Attribute\AsEntityListener;
use Symfony\Contracts\Cache\CacheInterface;

#[AsEntityListener(event: Events::preUpdate, entity: Location::class)]
#[AsEntityListener(event: Events::postUpdate, entity: Location::class)]
class LocationMapCacheInvalidator
{
    /** @var string[] */
    private array $pendingIds = [];

    public function __construct(private readonly CacheInterface $mapImagesCache) {}

    public function preUpdate(Location $location, PreUpdateEventArgs $args): void
    {
        if ($args->hasChangedField('latitude') || $args->hasChangedField('longitude')) {
            $this->pendingIds[] = $location->getId();
        }
    }

    public function postUpdate(Location $location, PostUpdateEventArgs $args): void
    {
        $id = $location->getId();
        if (!in_array($id, $this->pendingIds, true)) {
            return;
        }

        $this->mapImagesCache->delete('location_' . $id);
        $this->pendingIds = array_values(array_filter($this->pendingIds, fn($i) => $i !== $id));
    }
}
```

```php
namespace App\Controller;

use App\Service\Map\GeocoderInterface;
use App\Service\Map\MapImageProviderInterface;
use Doctrine\ORM\EntityManagerInterface;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;
use Symfony\Component\Security\Http\Attribute\IsGranted;
use Symfony\Contracts\Cache\CacheInterface;
use Symfony\Contracts\Cache\ItemInterface;

#[Route('/api/map')]
class MapController extends AbstractController
{
    private const IMAGE_WIDTH  = 800;
    private const IMAGE_HEIGHT = 288;
    private const IMAGE_ZOOM   = 14;

    public function __construct(
        private readonly GeocoderInterface $geocoder,
        private readonly MapImageProviderInterface $mapImageProvider,
        private readonly EntityManagerInterface $em,
        private readonly CacheInterface $mapImagesCache,
    ) {}

    #[Route('/geocode/suggest', methods: ['GET'])]
    #[IsGranted('ROLE_USER')]
    public function suggest(Request $request): Response
    {
        $query = trim((string) $request->query->get('q', ''));
        if ($query === '') {
            return $this->json(['suggestions' => []]);
        }

        $lat   = $request->query->has('lat') ? (float) $request->query->get('lat') : null;
        $lng   = $request->query->has('lng') ? (float) $request->query->get('lng') : null;
        $limit = max(1, min(10, (int) $request->query->get('limit', 6)));

        try {
            $results = $this->geocoder->suggest($query, $lat, $lng, $limit);
        } catch (\Throwable) {
            return $this->json(['error' => 'Geocoding service unavailable'], 503);
        }

        return $this->json([
            'suggestions' => array_map(fn($r) => [
                'label'   => $r->label,
                'lat'     => $r->lat,
                'lng'     => $r->lng,
                'street'  => $r->street,
                'zip'     => $r->zip,
                'city'    => $r->city,
                'country' => $r->country,
            ], $results),
        ]);
    }

    #[Route('/geocode/reverse', methods: ['GET'])]
    #[IsGranted('ROLE_USER')]
    public function reverse(Request $request): Response
    {
        $lat = (float) $request->query->get('lat', 0);
        $lng = (float) $request->query->get('lng', 0);

        try {
            $address = $this->geocoder->reverse($lng, $lat);
        } catch (\Throwable) {
            return $this->json(['error' => 'Geocoding service unavailable'], 503);
        }

        if ($address === null) {
            return $this->json(['error' => 'No address found'], 404);
        }

        return $this->json([
            'label'   => $address->label,
            'street'  => $address->street,
            'zip'     => $address->zip,
            'city'    => $address->city,
            'country' => $address->country,
        ]);
    }

}
```
